' ── Hires sheet event hook ────────────────────────────────────────────────
' This is NOT a module — paste it into the CODE PANE OF THE "Hires" SHEET
' (in the VBA editor, double-click "Hires" under Microsoft Excel Objects).
'
' On a hire-row edit it: stamps a HireId/audit on a new row; if the
' ContinuesFromHireId cell was the one edited, applies the continuation
' (completes the prior hire, copies contact/address/dates); then runs the
' double-booking / payment guard and, if it fails, undoes the edit.
'
' Single-owner editing assumed. Guarded against re-entrancy.

Private mBusy As Boolean

Private Sub Worksheet_Change(ByVal Target As Range)
    If mBusy Then Exit Sub

    Dim lo As ListObject
    On Error Resume Next
    Set lo = Me.ListObjects("HiresTable")
    On Error GoTo 0
    If lo Is Nothing Then Exit Sub
    If lo.DataBodyRange Is Nothing Then Exit Sub

    Dim hit As Range
    Set hit = Application.Intersect(Target, lo.DataBodyRange)
    If hit Is Nothing Then Exit Sub

    mBusy = True
    Application.EnableEvents = False
    On Error GoTo cleanup

    ' Which table column is the continuation link? (to detect a continuation edit)
    Dim contColAbs As Long
    contColAbs = lo.Range.Column + lo.ListColumns(BoxOfficeRules.CONTINUES_HEADER).Index - 1

    ' Process each affected row once.
    Dim seen As Object: Set seen = CreateObject("Scripting.Dictionary")
    Dim cell As Range
    For Each cell In hit.Cells
        Dim rowIdx As Long
        rowIdx = cell.Row - lo.DataBodyRange.Row + 1
        If Not seen.Exists(rowIdx) Then
            seen.Add rowIdx, True

            BoxOfficeRules.StampRow rowIdx

            ' Apply the continuation only if the link cell itself was edited in
            ' this change (so we don't re-complete the prior hire on every edit).
            Dim contEdited As Boolean: contEdited = False
            Dim c2 As Range
            For Each c2 In hit.Cells
                If c2.Row = cell.Row And c2.Column = contColAbs Then contEdited = True
            Next c2
            If contEdited Then BoxOfficeRules.ApplyContinuationForRow rowIdx

            Dim msg As String
            msg = BoxOfficeRules.ValidateHireRow(rowIdx)
            If msg <> "" Then
                Application.EnableEvents = True   ' Undo itself fires events
                mBusy = False
                Application.Undo
                MsgBox msg, vbExclamation, "Box Office — change rejected"
                Exit Sub
            End If
        End If
    Next cell

    ' Hires changed -> refresh the derived Terminals columns.
    On Error Resume Next
    RefreshTerminals.Refresh
    On Error GoTo 0

cleanup:
    Application.EnableEvents = True
    mBusy = False
End Sub
